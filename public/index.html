const express = require('express'); const axios = require('axios'); const app = express(); app.use(express.json()); app.use(express.static('public')); const PAYSTACK_SECRET_KEY = process.env.PAYSTACK_SECRET_KEY || 'sk_test_your_paystack_secret_key'; // Mock DB state for tables and bookings let bookings = []; const TOTAL_TABLES = 22; // 1. Get Available Tables app.get('/api/tables/available', (req, res) => { const { date, startTime, duration } = req.query; if (!date || !startTime || !duration) return res.status(400).json({ error: 'Missing parameters' }); const start = parseInt(startTime.split(':')[0]); const end = start + parseInt(duration); const bookedTableIds = bookings .filter(b => b.date === date && b.status === 'confirmed' && ((start >= b.start && start < b.end) || (end > b.start && end <= b.end))) .flatMap(b => b.tableIds); const availableTables = []; for (let i = 1; i <= TOTAL_TABLES; i++) { if (!bookedTableIds.includes(i)) { availableTables.push({ id: i, number: i, hourly_rate: 50 }); } } res.json(availableTables); }); // 2. Weekly Availability Grid Data app.get('/api/weekly-availability', (req, res) => { const { startDate } = req.query; const result = {}; for (let d = 0; d < 7; d++) { const currentDate = new Date(startDate); currentDate.setDate(currentDate.getDate() + d); const dateStr = currentDate.toISOString().split('T')[0]; result[dateStr] = {}; for (let hour = 11; hour <= 21; hour++) { const bookedCount = bookings .filter(b => b.date === dateStr && b.status === 'confirmed' && (hour >= b.start && hour < b.end)) .reduce((acc, b) => acc + b.tableIds.length, 0); result[dateStr][hour] = { available: Math.max(0, TOTAL_TABLES - bookedCount), total: TOTAL_TABLES }; } } res.json(result); }); // 3. Initialize Paystack Online Payment Transaction app.post('/api/bookings/initialize-payment', async (req, res) => { try { const { tableIds, date, startTime, durationHours, slot, userName, phone, email, totalPrice } = req.body; const start = parseInt(startTime.split(':')[0]); const end = start + parseInt(durationHours); // Paystack expects amount in minor currency unit (Cents / Kobo -> multiply by 100) const amountInCents = Math.round(parseFloat(totalPrice) * 100); const response = await axios.post( 'https://api.paystack.co/transaction/initialize', { email: email || `${phone}@cuecraft.com`, amount: amountInCents, currency: 'ZAR', callback_url: `${req.protocol}://${req.get('host')}/payment-success.html`, metadata: { tableIds, date, startTime, start, end, durationHours, slot, userName, phone } }, { headers: { Authorization: `Bearer ${PAYSTACK_SECRET_KEY}`, 'Content-Type': 'application/json' } } ); res.json({ authorization_url: response.data.data.authorization_url, reference: response.data.data.reference }); } catch (err) { console.error('Paystack initialization error:', err.response?.data || err.message); res.status(500).json({ error: 'Failed to initialize online payment.' }); } }); // 4. Paystack Webhook / Verification Endpoint app.post('/api/bookings/verify-payment', async (req, res) => { const { reference } = req.body; try { const response = await axios.get(`https://api.paystack.co/transaction/verify/${reference}`, { headers: { Authorization: `Bearer ${PAYSTACK_SECRET_KEY}` } }); const data = response.data.data; if (data.status === 'success') { const meta = data.metadata; // Confirm reservation in DB bookings.push({ reference, tableIds: meta.tableIds, date: meta.date, start: meta.start, end: meta.end, userName: meta.userName, phone: meta.phone, status: 'confirmed' }); return res.json({ success: true, message: 'Reservation confirmed!' }); } res.status(400).json({ error: 'Payment verification failed.' }); } catch (err) { res.status(500).json({ error: 'Error verifying payment.' }); } }); app.listen(3000, () => console.log('CueCraft server running on port 3000'));
